# ==============================================================
# Distribution des notes de liking (échelle 1-9)
# Jeu de données : Fried Mozzarella Cheese Sticks (Sensometric Society)
# ==============================================================

# Passage en format long
liking_long <- consumers %>%
  select(Panelist, `Prod ID`, Overall, Appearance, Flavor, Texture) %>%
  pivot_longer(
    cols = c(Overall, Appearance, Flavor, Texture),
    names_to = "Dimension",
    values_to = "Note"
  ) %>%
  mutate(
    Dimension = factor(
      Dimension,
      levels = c("Overall", "Appearance", "Flavor", "Texture"),
      labels = c("Appréciation globale", "Apparence", "Saveur", "Texture")
    ),
    Note = factor(Note, levels = 1:9)
  )

# Distribution en pourcentage, par dimension et par note
dist_notes <- liking_long %>%
  count(Dimension, Note, name = "n") %>%
  complete(Dimension, Note, fill = list(n = 0)) %>%
  group_by(Dimension) %>%
  mutate(pct = n / sum(n)) %>%
  ungroup()

# Moyenne par dimension, pour la ligne repère
moyennes <- liking_long %>%
  mutate(Note_num = as.numeric(as.character(Note))) %>%
  group_by(Dimension) %>%
  summarise(moyenne = mean(Note_num), .groups = "drop")

# Palette par dimension
palette_dim <- c(
  "Appréciation globale" = "#2a78d6",
  "Apparence"             = "#1baf7a",
  "Saveur"               = "#e34948",
  "Texture"               = "#4a3aa7"
)

# Graphique
p <- ggplot(dist_notes, aes(x = Note, y = pct, fill = Dimension)) +
  geom_col(width = 0.72, show.legend = FALSE) +
  geom_vline(
    data = moyennes,
    aes(xintercept = moyenne),
    linetype = "dashed", linewidth = 0.6, colour = "grey40"
  ) +
  geom_text(
    data = moyennes,
    aes(x = moyenne, y = Inf, label = paste0("moy = ", round(moyenne, 2))),
    inherit.aes = FALSE, vjust = 1.6, hjust = -0.08, size = 3.1,
    colour = "grey40", fontface = "bold"
  ) +
  facet_wrap(~ Dimension, nrow = 2, scales = "free_x") +
  scale_fill_manual(values = palette_dim) +
  scale_x_discrete(drop = FALSE) +
  scale_y_continuous(
    labels = percent_format(accuracy = 1),
    expand = expansion(mult = c(0, 0.18))
  ) +
  labs(
    title = "Distribution des notes de liking (échelle 1-9)",
    subtitle = "161 consommateurs x 8 bâtonnets de mozzarella panée",
    x = "Note (1 = je déteste -- 9 = j'adore)",
    y = NULL,
    caption = "Source : jeu de données Fried Mozzarella Cheese Sticks, Sensometric Society"
  ) +
  theme_minimal(base_size = 12) +
  theme(
    plot.title = element_text(face = "bold", size = 15),
    plot.subtitle = element_text(colour = "grey40", size = 10, margin = margin(b = 10)),
    plot.caption = element_text(colour = "grey55", size = 8, hjust = 0),
    strip.text = element_text(face = "bold", hjust = 0, size = 11, colour = "grey10"),
    strip.background = element_blank(),
    panel.spacing = unit(1.1, "lines"),
    panel.grid.minor = element_blank(),
    panel.grid.major.x = element_blank(),
    panel.grid.major.y = element_line(colour = "grey90", linewidth = 0.4),
    axis.text = element_text(colour = "grey45"),

moyennes_produit <- consumers %>%
  group_by(`Prod ID`) %>%
  summarise(
    Overall    = mean(Overall),
    Appearance = mean(Appearance),
    Flavor     = mean(Flavor),
    Texture    = mean(Texture),
    .groups = "drop"
  ) %>%
  pivot_longer(cols = c(Overall, Appearance, Flavor, Texture),
               names_to = "Dimension", values_to = "moyenne") %>%
  mutate(Dimension = factor(Dimension,
           levels = c("Overall", "Appearance", "Flavor", "Texture"),
           labels = c("Appréciation globale", "Apparence", "Flaveur", "Texture")))

ordre_produits <- moyennes_produit %>%
  filter(Dimension == "Appréciation globale") %>%

  cat("=============================================\n")

  form <- as.formula(paste(var, "~ Prod.ID"))

  # construction du modèle 
  res <- LinearModel(form, data = dta_cons)

  # affichage de la sortie du modèle
  print(res)

  # test post-hoc
  cat("\n--- Test de comparaison des moyennes (meansComp) ---\n")
  print(meansComp(res, ~ Prod.ID))
}

# ==============================================================
# Étendue et pouvoir discriminant des 37 descripteurs du jury
# Fichier : mozza_panel_descriptors.csv (8 produits x 37 descripteurs)
# ==============================================================

# 1. Table de synthèse : un descripteur par ligne
synth_descripteurs <- jury %>%
  select(-Prod.ID) %>%
  pivot_longer(everything(), names_to = "descripteur", values_to = "valeur") %>%
  group_by(descripteur) %>%
  summarise(
    min        = min(valeur),
    max        = max(valeur),
    etendue    = max - min,
    moyenne    = mean(valeur),
    ecart_type = sd(valeur),
    cv_pct     = 100 * ecart_type / moyenne,
    .groups = "drop"
  ) %>%
  arrange(desc(cv_pct))

# 2. Lollipop horizontal : CV% trié du plus stable au plus discriminant
seuil_cv <- 10  # seuil indicatif

ggplot(synth_descripteurs,
       aes(x = fct_reorder(descripteur, cv_pct), y = cv_pct)) +
  geom_segment(aes(xend = descripteur, y = 0, yend = cv_pct),
               colour = "grey75", linewidth = 0.6) +
  geom_point(aes(colour = cv_pct), size = 2.6) +
  geom_hline(yintercept = seuil_cv, linetype = "dashed",
             colour = "grey40", linewidth = 0.5) +
  annotate("text", x = nrow(synth_descripteurs) - 1, y = seuil_cv,
           label = paste0("seuil ", seuil_cv, " %"),
           vjust = -0.6, hjust = 0, size = 3, colour = "grey40") +
  coord_flip() +
  scale_colour_gradient(low = "#c9d9f2", high = "#1f4e9c", guide = "none") +
  scale_y_continuous(labels = label_percent(scale = 1, accuracy = 1),
                     expand = expansion(mult = c(0, 0.08))) +
  labs(
    title = "Pouvoir discriminant des 37 descripteurs du jury",
    subtitle = "Coefficient de variation (écart-type / moyenne) entre les 8 produits",
    x = NULL, y = "Coefficient de variation",
    caption = "Source : mozza_panel_descriptors.csv -- 8 produits x 37 descripteurs"
  ) +
  theme_minimal(base_size = 11) +
  theme(
    plot.title = element_text(face = "bold", size = 14),
    plot.subtitle = element_text(colour = "grey40", size = 9.5, margin = margin(b = 10)),

Analyse Factorielle Multiple (AFM)

Préparation des données pour l’AFM

On calcule les moyennes des 4 dimensions d’appréciation consommateurs par produit, puis on fusionne ces moyennes avec les 37 descripteurs du jury.

Code 
# Moyennes consommateurs par produit
dta_cons_m <- dta_cons %>%
  group_by(Prod.ID) %>%
  summarise(
    cons_overall    = mean(Overall, na.rm = TRUE),
    cons_texture    = mean(Texture, na.rm = TRUE),
    cons_appearance = mean(Appearance, na.rm = TRUE),
    cons_flavor     = mean(Flavor, na.rm = TRUE)
  )

# Fusion avec le panel jury
dta_afm <- inner_join(dta_cons_m, dta_panel, by = "Prod.ID") %>%
  column_to_rownames("Prod.ID")
